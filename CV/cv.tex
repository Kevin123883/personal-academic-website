%!TEX encoding = UTF-8 Unicode
% CV — plain one-column layout using resume.cls.
%
% This file is the CV's "skin": page geometry, list spacing and which sections
% appear (and in what order). The section files under cv/ are GENERATED from
% data/*.json by scripts/build-cv.mjs — edit the JSON and run `npm run cv`.

\documentclass{resume} % Use the custom resume.cls style

\usepackage[left=0.75in,top=0.6in,right=0.75in,bottom=0.6in]{geometry} % Document margins
\usepackage{enumitem}
\usepackage[hidelinks]{hyperref}

% Tight bullets (replaces the per-item \vspace{-0.3cm} hack in the original
% template). parskip would otherwise add a blank line between every item and
% between a heading line and its list; the negative topsep cancels that, and
% the hook puts the normal paragraph gap back below the list.
\setlist[itemize]{topsep=-\parskip, itemsep=0pt, parsep=0pt, partopsep=0pt, leftmargin=1.5em}
\AddToHook{env/itemize/after}{\vspace{\parskip}}

%-------------------------------------------------------------------------------
%	PERSONAL INFORMATION — generated from data/about.json (\name, \address)
%-------------------------------------------------------------------------------
\input{cv/_personal.tex}

\begin{document}

%-------------------------------------------------------------------------------
%	CONTENT — a section whose data is empty generates an empty file and is
%	skipped automatically. Comment out a line to drop that section.
%-------------------------------------------------------------------------------
\input{cv/education.tex}
\input{cv/interests.tex}
\input{cv/professional.tex}
\input{cv/publications.tex}
\input{cv/honors.tex}
\input{cv/presentation.tex}
\input{cv/teaching.tex}
\input{cv/mentoring.tex}
\input{cv/extracurricular.tex}

\end{document}
